$\displaystyle
\begin{aligned}
\mathbb{P}^{n}_{K}
&=\left\{x\in\mathbb{R}^{n}\mid\lVert x\rVert^{2}<-1/K\right\},\\[4pt]
\mathbb{L}^{n}_{K}
&=\left\{x=\left[x_t,x_s^{\top}\right]^{\top}\in\mathbb{R}^{n+1}
\mid\left\langle x,x\right\rangle_{\mathcal{L}}=1/K,\ x_t>0\right\},\\[4pt]
\left\langle x,y\right\rangle_{\mathcal{L}}
&=-x_ty_t+\left\langle x_s,y_s\right\rangle,
\qquad
\mathcal{H}^{n}_{K}\in\left\{\mathbb{P}^{n}_{K},\mathbb{L}^{n}_{K}\right\}.
\end{aligned}
